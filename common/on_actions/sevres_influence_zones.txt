				TUR = {

					}
					give_resource_rights = {
						receiver = ITA
						state = 345

						receiver = ITA
						state = 346
					}

						receiver = FRA
						state = 344
					}

						receiver = FRA
						state = 348
					}

						state = 349
					}
				}

				KUR = {

						state = 350
					}

				}
			}
		}
	}
}
