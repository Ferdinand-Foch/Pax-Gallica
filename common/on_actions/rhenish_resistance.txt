# Initialize without setting resistance; update only the four listed states each day.
on_actions = {
	on_startup = {
		effect = {
			FRA = {
				set_country_flag = rhenish_notification_suppressed
				rhenish_resistance_update = yes
				clr_country_flag = rhenish_notification_suppressed
			}
		}
	}
	on_daily_FRA = {
		effect = { rhenish_resistance_update = yes }
	}
}
