# Assigns the unchanged vanilla generic tree to the listed countries at the 1936 start.
on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { date < 1936.1.2 }
				FRA = {
					load_focus_tree = { tree = generic_focus }
				}
				GER = {
					load_focus_tree = { tree = generic_focus }
				}
				AUS = {
					load_focus_tree = { tree = generic_focus }
				}
				BEL = {
					load_focus_tree = { tree = generic_focus }
				}
				HOL = {
					load_focus_tree = { tree = generic_focus }
				}
			}
		}
	}
}
