# Local tiers and French national tier maintenance. Native resistance is only read.

rhenish_resistance_clear = {
	if = {
		limit = { has_dynamic_modifier = rhenish_resistance_light }
		remove_dynamic_modifier = { modifier = rhenish_resistance_light }
	}
	if = {
		limit = { has_dynamic_modifier = rhenish_resistance_general }
		remove_dynamic_modifier = { modifier = rhenish_resistance_general }
	}
	if = {
		limit = { has_dynamic_modifier = rhenish_resistance_sabotage }
		remove_dynamic_modifier = { modifier = rhenish_resistance_sabotage }
	}
	if = {
		limit = { has_dynamic_modifier = rhenish_resistance_armed }
		remove_dynamic_modifier = { modifier = rhenish_resistance_armed }
	}
	if = {
		limit = { has_dynamic_modifier = rhenish_resistance_uprisings }
		remove_dynamic_modifier = { modifier = rhenish_resistance_uprisings }
	}
}

rhenish_resistance_update_state = {
	if = {
		limit = { rhenish_resistance_eligible = yes }
		if = {
			limit = { check_variable = { var = resistance value = constant:rhenish_resistance.armed_ceiling compare = greater_than } }
			if = {
				limit = { NOT = { has_dynamic_modifier = rhenish_resistance_uprisings } }
				rhenish_resistance_clear = yes
				add_dynamic_modifier = { modifier = rhenish_resistance_uprisings }
			}
		}
		else_if = {
			limit = { check_variable = { var = resistance value = constant:rhenish_resistance.sabotage_ceiling compare = greater_than } }
			if = {
				limit = { NOT = { has_dynamic_modifier = rhenish_resistance_armed } }
				rhenish_resistance_clear = yes
				add_dynamic_modifier = { modifier = rhenish_resistance_armed }
			}
		}
		else_if = {
			limit = { check_variable = { var = resistance value = constant:rhenish_resistance.general_ceiling compare = greater_than } }
			if = {
				limit = { NOT = { has_dynamic_modifier = rhenish_resistance_sabotage } }
				rhenish_resistance_clear = yes
				add_dynamic_modifier = { modifier = rhenish_resistance_sabotage }
			}
		}
		else_if = {
			limit = { check_variable = { var = resistance value = constant:rhenish_resistance.light_ceiling compare = greater_than } }
			if = {
				limit = { NOT = { has_dynamic_modifier = rhenish_resistance_general } }
				rhenish_resistance_clear = yes
				add_dynamic_modifier = { modifier = rhenish_resistance_general }
			}
		}
		else = {
			if = {
				limit = { NOT = { has_dynamic_modifier = rhenish_resistance_light } }
				rhenish_resistance_clear = yes
				add_dynamic_modifier = { modifier = rhenish_resistance_light }
			}
		}
	}
	else = { rhenish_resistance_clear = yes }
}

# State scope. The accumulator is initialized by the calling country effect.
rhenish_resistance_update_and_accumulate = {
	rhenish_resistance_update_state = yes
	if = {
		limit = {
			rhenish_resistance_eligible = yes
			check_variable = { var = resistance value = rhenish_max_resistance compare = greater_than }
		}
		set_temp_variable = { rhenish_max_resistance = resistance }
		save_event_target_as = rhenish_determining_state
	}
}

# FRA scope. Each idea's on_remove also removes its four GER -> FRA intel tokens.
rhenish_resistance_clear_national = {
	if = {
		limit = { has_idea = rhenish_occupation }
		remove_ideas = rhenish_occupation
	}
	if = {
		limit = { has_idea = rhenish_concern }
		remove_ideas = rhenish_concern
	}
	if = {
		limit = { has_idea = rhenish_crisis }
		remove_ideas = rhenish_crisis
	}
	if = {
		limit = { has_idea = rhenish_insurrection }
		remove_ideas = rhenish_insurrection
	}
	if = {
		limit = { has_idea = rhenish_uprising }
		remove_ideas = rhenish_uprising
	}
}

# FRA scope. Strict descending comparisons preserve inclusive upper boundaries.
rhenish_resistance_update_national = {
	if = {
		limit = { check_variable = { var = rhenish_max_resistance value = constant:rhenish_resistance.no_eligible_resistance compare = equals } }
		rhenish_resistance_clear_national = yes
	}
	else_if = {
		limit = { check_variable = { var = rhenish_max_resistance value = constant:rhenish_resistance.armed_ceiling compare = greater_than } }
		if = {
			limit = { NOT = { has_idea = rhenish_uprising } }
			rhenish_resistance_clear_national = yes
			add_ideas = rhenish_uprising
		}
	}
	else_if = {
		limit = { check_variable = { var = rhenish_max_resistance value = constant:rhenish_resistance.sabotage_ceiling compare = greater_than } }
		if = {
			limit = { NOT = { has_idea = rhenish_insurrection } }
			rhenish_resistance_clear_national = yes
			add_ideas = rhenish_insurrection
		}
	}
	else_if = {
		limit = { check_variable = { var = rhenish_max_resistance value = constant:rhenish_resistance.general_ceiling compare = greater_than } }
		if = {
			limit = { NOT = { has_idea = rhenish_crisis } }
			rhenish_resistance_clear_national = yes
			add_ideas = rhenish_crisis
		}
	}
	else_if = {
		limit = { check_variable = { var = rhenish_max_resistance value = constant:rhenish_resistance.light_ceiling compare = greater_than } }
		if = {
			limit = { NOT = { has_idea = rhenish_concern } }
			rhenish_resistance_clear_national = yes
			add_ideas = rhenish_concern
		}
	}
	else = {
		if = {
			limit = { NOT = { has_idea = rhenish_occupation } }
			rhenish_resistance_clear_national = yes
			add_ideas = rhenish_occupation
		}
	}
}

# FRA scope: one shared four-state pass at startup and in the existing daily hook.
rhenish_resistance_update = {
	rhenish_notification_read_tier = yes
	set_temp_variable = { rhenish_previous_tier = rhenish_notification_tier }
	set_temp_variable = { rhenish_max_resistance = constant:rhenish_resistance.no_eligible_resistance }
	42 = { rhenish_resistance_update_and_accumulate = yes }
	1082 = { rhenish_resistance_update_and_accumulate = yes }
	1083 = { rhenish_resistance_update_and_accumulate = yes }
	1084 = { rhenish_resistance_update_and_accumulate = yes }
	rhenish_resistance_update_national = yes
	rhenish_resistance_notify_change = yes
}

# Notification metadata only; existing ideas remain the source of truth.
rhenish_notification_read_tier = {
	set_temp_variable = { rhenish_notification_tier = 0 }
	if = {
		limit = { has_idea = rhenish_occupation }
		set_temp_variable = { rhenish_notification_tier = 1 }
	}
	if = {
		limit = { has_idea = rhenish_concern }
		set_temp_variable = { rhenish_notification_tier = 2 }
	}
	if = {
		limit = { has_idea = rhenish_crisis }
		set_temp_variable = { rhenish_notification_tier = 3 }
	}
	if = {
		limit = { has_idea = rhenish_insurrection }
		set_temp_variable = { rhenish_notification_tier = 4 }
	}
	if = {
		limit = { has_idea = rhenish_uprising }
		set_temp_variable = { rhenish_notification_tier = 5 }
	}
}

# One informational event after the complete national update.
rhenish_resistance_notify_change = {
	rhenish_notification_read_tier = yes
	if = {
		limit = {
			tag = FRA
			is_ai = no
			NOT = { has_country_flag = rhenish_notification_suppressed }
			check_variable = { rhenish_previous_tier > 0 }
			check_variable = { rhenish_notification_tier > 0 }
			NOT = { check_variable = { var = rhenish_previous_tier value = rhenish_notification_tier compare = equals } }
		}
		set_variable = { rhenish_notification_old = rhenish_previous_tier }
		set_variable = { rhenish_notification_new = rhenish_notification_tier }
		set_variable = { rhenish_notification_resistance = rhenish_max_resistance }
		if = {
			limit = { check_variable = { var = rhenish_notification_tier value = rhenish_previous_tier compare = greater_than } }
			country_event = { id = rhenish_notification.1 }
		}
		else = { country_event = { id = rhenish_notification.2 } }
	}
}
